==========================================
Building Contract IQ Docker Images
==========================================

IMPORTANT: Start Docker Desktop before running these commands!

Step 1: Build Backend Image
----------------------------
cd packages\backend
docker build -t contract-iq-backend:latest .
cd ..\..

Step 2: Build Frontend Image
-----------------------------
cd packages\frontend

# Read environment variables from .env.local
$env = Get-Content .env.local | Where-Object { $_ -match "=" }
$supabaseUrl = ($env | Where-Object { $_ -match "NEXT_PUBLIC_SUPABASE_URL" }) -replace ".*=", ""
$supabaseKey = ($env | Where-Object { $_ -match "NEXT_PUBLIC_SUPABASE_ANON_KEY" }) -replace ".*=", ""
$backendUrl = ($env | Where-Object { $_ -match "NEXT_PUBLIC_BACKEND_URL" }) -replace ".*=", ""

# If backend URL is not set, use default
if (!$backendUrl) { $backendUrl = "http://localhost:8000" }

docker build --build-arg NEXT_PUBLIC_SUPABASE_URL="$supabaseUrl" --build-arg NEXT_PUBLIC_SUPABASE_ANON_KEY="$supabaseKey" --build-arg NEXT_PUBLIC_BACKEND_URL="$backendUrl" -t contract-iq-frontend:latest .

cd ..\..

Step 3: Verify Images
----------------------
docker images | findstr contract-iq

==========================================
Done! Images are ready.
==========================================
